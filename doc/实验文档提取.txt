===== GPS模块实验 =====
GPS模块实验
实验目的
1、使用 SIM8262E-M2 5G 模组内置的 GNSS 功能，通过 AT 指令启动 GPS，在终端每秒打印经纬度、卫星数、速度等定位信息。
2、掌握 NMEA 0183 报文的串口流式解析方法：strtok 按 \n 拆包、strstr 匹配报文类型、parse_nmea_field 按下标抽取逗号分隔字段。
实验内容
1、硬件接线：SIM8262E-M2 5G 模组 AT 通道接 /dev/ttyUSB2，GNSS 数据通道接 /dev/ttyUSB1，波特率统一 115200。
2、串口初始化：基于 termios 实现 8N1 无校验、无硬件流控、原始模式（raw mode）的串口配置函数 gps_init()。
3、GNSS 使能：通过 AT 通道发送 AT+CGPS=1 和 AT+CGPSOUT=64 两条指令，启动 SIM8262E-M2 内置 GNSS 并配置 NMEA 输出内容。
4、NMEA 报文规则学习：对照 gps讲解.docx，逐条理解 $ 开头、逗号分隔、* 校验、\r\n 结尾的 5 条硬性规则。
5、三类报文解析：分别实现 parse_gpgga（时间/经纬度/定位状态/卫星数/HDOP/海拔）、parse_gprmc（速度/航向/日期）、parse_gpgsv（总可见卫星数）三个解析函数。
6、主循环分发：在 gps_process 中用 read 批量读取缓冲区，strtok 按行拆分，再用 strstr 判断报文类型并分发到对应解析函数。
7、结果打印与验证：每秒调用 gps_print_info 输出一条格式化 GPS 信息，对比卫星实际分布与定位精度。
实验仪器
1、OrangePi 5 嵌入式开发板一台。
2、SIM8262E-M2 5G 通信模组一块（高通骁龙 X62 芯片，3GPP R16，集成多星座 GNSS），配套 M.2 转 USB 扩展板。
3、GNSS 有源天线一根（接 SIM8262E-M2 的 ANT3 接口，注意有字一面朝下）。
4、电脑一台，使用 MobaXterm 或其他 SSH 工具远程连接香橙派进行开发。
5、arm-linux-gnueabihf-gcc / aarch64-linux-gnu-gcc 交叉编译工具链，或香橙派本机 gcc 编译环境。
6、室外开阔场地或窗边（用于接收卫星信号，室内通常无法定位）。
实验原理
SIM8262E-M2 启动 GNSS 后，通过数据串口持续输出 NMEA 0183 报文。每条报文通用格式如下：
$报文标识,字段1,字段2,字段3,...,最后字段*校验码\r\n
代码硬性依赖的 5 条规则：
1、每条报文以 $ 开头，\n 换行分隔，strtok(buf, "\n") 拆分单条数据包。
2、字段用英文逗号 , 分割，字段下标从 0 开始。
3、解析只读取 $ ~ * 校验符之间内容，parse_nmea_field 遇到 * 停止遍历。
4、串口流式输出，多包混杂在缓冲区，gps_process 一次性读取后分行解析。
5、常见报文类型：$GPGGA（定位信息）、$GPRMC（推荐最小定位）、$GPGSV（可见卫星）。
Eg. 一条 $GPGGA 报文示例：
$GPGGA,152230.000,3931.4160,N,11642.0720,E,1,08,1.02,22.0,M,,M,,*78\r\n
含义：UTC 15:22:30，纬度 39°31.4160′N，经度 116°42.0720′E，已定位，8 颗卫星，HDOP 1.02，海拔 22.0m。程序通过 parse_nmea_field 按逗号下标抽取各字段存入 GPS_Data 结构体，最后由 gps_print_info 格式化打印到终端。

int gps_init(const char *dev_path, int baudrate)
{
    int fd = open(dev_path, O_RDWR | O_NOCTTY);
    if (fd &lt; 0) return -1;
    struct termios opt;
    tcgetattr(fd, &amp;opt);
    cfsetispeed(&amp;opt, baudrate);
    cfsetospeed(&amp;opt, baudrate);
    opt.c_cflag   = CREAD | CLOCAL;
    opt.c_cflag  |= CS8;
    opt.c_cflag  &amp;= ~PARENB;
    opt.c_cflag  &amp;= ~CSTOPB;
    opt.c_iflag  &amp;= ~(IXON | IXOFF | IXANY | ICRNL);
    opt.c_lflag  &amp;= ~(ICANON | ECHO | ECHOE | ISIG);
    opt.c_oflag  &amp;= ~OPOST;
    tcflush(fd, TCIFLUSH);
    tcsetattr(fd, TCSANOW, &amp;opt);
    return fd;
}

int gps_enable_at(const char *at_dev, int baudrate)
{
    int fd = gps_init(at_dev, baudrate);
    if (fd &lt; 0) return -1;
    const char *cmd1 = "AT+CGPS=1\r\n";
    const char *cmd2 = "AT+CGPSOUT=64\r\n";
    write(fd, cmd1, strlen(cmd1)); usleep(300000);
    write(fd, cmd2, strlen(cmd2)); usleep(300000);
    close(fd);
    return 0;
}
四：通用字段抽取 — parse_nmea_field()
该函数是三个解析函数的公用基础：在逗号分隔字符串中找到第 target_field 个字段，复制到 result，遇到 * 结束（避免把校验符拷进数据）。
对应代码片段（提取自素材图5 函数声明，结合素材图3/4 使用方式还原）：
static void parse_nmea_field(const char *line, int target_field,
                              char *result, int max_len)
{
    int i, cur = 0, ri = 0;
    for (i = 0; line[i] &amp;&amp; cur &lt;= target_field; i++) {
        if (line[i] == ',') { cur++; continue; }
        if (line[i] == '*') break;
        if (cur == target_field &amp;&amp; ri &lt; max_len - 1) {
            result[ri++] = line[i];
        }
    }
    result[ri] = '\0';
}

static void parse_gpgga(const char *line, GPS_Data *gps)
{
    // GPGGA 数据包 — 提取 1~5 字段
    parse_nmea_field(line, 1, gps-&gt;utc_time, sizeof(gps-&gt;utc_time));
    parse_nmea_field(line, 2, gps-&gt;latitude, sizeof(gps-&gt;latitude));
    parse_nmea_field(line, 3, &amp;gps-&gt;lat_dir, 1);
    parse_nmea_field(line, 4, gps-&gt;longitude, sizeof(gps-&gt;longitude));
    parse_nmea_field(line, 5, &amp;gps-&gt;lon_dir, 1);
    // 6~9 字段：定位状态 + 卫星数 + HDOP + 海拔
    char status[2] = {0};
    parse_nmea_field(line, 6, status, sizeof(status));
    gps-&gt;status = status[0];
    parse_nmea_field(line, 7, gps-&gt;satellites, sizeof(gps-&gt;satellites));
    parse_nmea_field(line, 8, gps-&gt;hdop, sizeof(gps-&gt;hdop));
    parse_nmea_field(line, 9, gps-&gt;altitude, sizeof(gps-&gt;altitude));
}

static void parse_gpgsv(const char *line, GPS_Data *gps)
{

    char total[4] = {0};

对应代码片段（提取自素材图3）：
void gps_process(int fd, GPS_Data *gps)
{
    char buf[2048];
    memset(buf, 0, sizeof(buf));
    int len = read(fd, buf, sizeof(buf) - 1);
    // 只要读到任意数据 = 有 GPS 模块
    if (len &gt; 0) gps_module_detected = 1;
    if (len &lt;= 0) return;
    char *line = strtok(buf, "\n");
    while (line != NULL) {
        if (strstr(line, "$GPGGA")) parse_gpgga(line, gps);
        if (strstr(line, "$GPRMC")) parse_gprmc(line, gps);
        if (strstr(line, "$GPGSV")) parse_gpgsv(line, gps);
        line = strtok(NULL, "\n");
    }
}
九：格式化输出 — gps_print_info()
工程化打印规则（提取自素材图7 注释）：
1、有模块时只打印 GPS 信息，绝不提示未检测到。
2、无模块时不打印任何东西（避免刷屏干扰其他线程输出）。
对应代码片段（提取自素材图7）：
void gps_print_info(GPS_Data *gps)
{
    // 规则：
    // 1. 有模块时只打印 GPS 信息，绝不提示未检测到
    // 2. 无模块时不打印任何东西
    if (gps_module_detected) {
        printf("\n===== GPS INFO =====\n");
        printf("Time: %s | Status: %c | Sats: %s/%d\n",
               gps-&gt;utc_time, gps-&gt;status,
               gps-&gt;satellites, gps-&gt;total_satellites);
        printf("Lat: %s %c\n", gps-&gt;latitude, gps-&gt;lat_dir);
        printf("Lon: %s %c\n", gps-&gt;longitude, gps-&gt;lon_dir);
        printf("Alt: %s m | HDOP: %s\n", gps-&gt;altitude, gps-&gt;hdop);
        printf("Speed: %s kn | Course: %s deg\n", gps-&gt;speed, gps-&gt;course);
        printf("Date: %s\n", gps-&gt;date);
    }
}
十：主函数入口 — main()
main 函数整合所有步骤：先通过 AT 通道启动 GPS（sleep(1) 等冷启动），再打开 GPS 数据通道，创建一个 GPS_Data 实例，进入无限循环，每秒 process+print 一次。
对应代码片段（提取自素材图1）：
int main()
{
    // 开启 GPS
    gps_enable_at(AT_DEV, GPS_BAUD);
    sleep(1);
    int fd = gps_init(GPS_DEV, GPS_BAUD);
    GPS_Data gps = {0};
    printf("GPS started print every 1 second\n\n");
    while (1) {
        gps_process(fd, &amp;gps);
        gps_print_info(&amp;gps);
        sleep(1);
    }
    return 0;
}
实验结果

实验结果分析
1、AT 指令启动延迟分析：SIM8262E-M2 处理 AT+CGPS=1 需要切换射频链路并启动内部 GNSS 固件流程，两条 AT 指令之间必须加 usleep(300000) 等待模组处理完成。若不加延时，第二条 AT+CGPSOUT=64 可能在模组尚未就绪时写入而被丢弃，导致 GNSS 启动失败、数据通道无 NMEA 输出。300ms 是工程经验值，部分模组冷启动可能需要更长。
2、双串口架构的必要性：SIM8262E-M2 通过 USB 枚举出多个独立串口，AT 通道（/dev/ttyUSB2）用于发送控制指令，数据通道（/dev/ttyUSB1）专用于接收 NMEA 字节流。两者分离避免了 AT 指令响应与 NMEA 数据混杂在同一缓冲区导致的解析困难，gps_enable_at 发完指令即 close AT 通道，后续 gps_process 只专注数据通道读取。
3、strtok + strstr 流式解析的健壮性：串口缓冲区中多条 NMEA 报文混杂，strtok(buf, "\n") 按换行符拆分成独立报文，再用 strstr(line, "$GPGGA") 判断报文类型分发。只传 "\n" 而非 "\r\n" 是因为 strtok 分隔符参数中每个字符都当独立分隔符，\r 会留在行尾但不影响 strstr 的前向匹配，兼容 \r\n 和 \n 两种结尾的模组。
4、parse_nmea_field 边界安全设计：该函数按下标抽取逗号分隔字段，遇 * 停止遍历（不把校验码拷进数据），并用 max_len 限制写入长度防止缓冲区溢出。定位状态字段（下标6）只有一个字符 A/V，先读入临时 char[2] 再取 [0] 赋值给 char status，避免了直接写入单字符变量时 \0 越界覆盖相邻内存。
5、静默打印策略的工程意义：gps_module_detected 标志仅在 len&gt;0 时置 1 且从不清零，gps_print_info 据此决定是否输出。有模块时只打印 GPS 信息绝不提示"未检测到"，无模块时完全静默。这是因为 GPS 线程与小车其他线程（IMU、视觉、控制）共享终端，持续打印"No GPS"会淹没关键调试信息。
6、定位精度与卫星数的关系：实验中 satellites（GPGGA 下标7）为当前用于定位的卫星数，total_satellites（GPGSV 下标3）为接收机搜到的总可见卫星数。两者比值越高（≥60%）说明信号质量越好、定位精度越高。室内或遮挡环境下 satellites 可能为 0 且 status=V（未定位），移至室外开阔处后通常 30 秒~2 分钟内完成冷启动定位。

===== IMU传感器实验 =====
IMU传感器实验
实验目的
1、了解 JY901S 九轴姿态传感器的硬件组成：三轴加速度计、三轴陀螺仪、三轴磁力计，理解各自测量的物理量。
2、掌握 Linux 下 UART 串口通信编程（termios 配置）与 JY901S 数据帧格式：0x55 帧头 + CID + 8 字节数据 + 累加校验和，共 11 字节。
3、掌握按 CID 区分数据类型（0x51 加速度 / 0x52 陀螺仪 / 0x53 姿态角 / 0x54 磁力计）的解析方法。
4、掌握 16 位小端数据拼接（低字节在前）与原始值转物理量的换算方法。
5、理解状态机式串口组帧原理与配置指令帧 0xFF 0xAA 的发送方法。
实验内容
1、连接 JY901S 的 VCC、GND、TX、RX 四线到开发板串口，确认串口设备节点（如 /dev/ttyS5）。

实验原理
1、九轴传感器组成与欧拉角含义
JY901S 内部集成三个三轴传感器共九个测量轴：三轴加速度计（测量 X/Y/Z 方向重力加速度，单位 g）、三轴陀螺仪（测量角速度，单位 °/s）、三轴磁力计（测量地磁场强度，用于判断绝对航向角）。传感器内部集成姿态解算算法，直接通过串口输出四类数据。姿态角采用欧拉角表示：Roll 横滚角（绕 X 轴，正值右倾）、Pitch 俯仰角（绕 Y 轴，正值上仰）、Yaw 航向角（绕 Z 轴，九轴模式下为绝对航向）。
2、UART 串口通信规则与数据帧格式
JY901S 通过 TX/RX 串口线与主控通信，主控使用 termios API 配置 8N1。传感器持续以字节流主动发送 11 字节数据帧，主控按字节读取并自行组帧。
数据帧结构如下：
字节1   字节2   字节3~字节10          字节11
0x55     CID    DATA[0]~DATA[7]       SUM
其中 CID 为数据类型标识：0x51 加速度、0x52 陀螺仪、0x53 姿态角、0x54 磁力计。DATA 段 8 字节为 3 组 16 位小端数据（低字节在前）+ 2 字节保留。SUM 为前 10 字节累加校验和。组帧采用状态机：单字节读取 → 找到 0x55 帧头 → 凑齐 11 字节 → 校验和通过 → 解析。
3、配置指令帧与校验和计算
主控向 JY901S 发送配置指令帧，格式为 5 字节：0xFF 0xAA + 寄存器地址 + 数据值 + 0x00。例如开启全量程模式指令：0xFF 0xAA 0x23 0x01 0x00。校验和函数对 11 字节帧的前 10 字节累加求和（无符号 char 溢出回绕），与第 11 字节比对验证数据完整性。
4、原始值转物理量
解析得到的 int16 原始值需按传感器量程换算为实际物理量。加速度量程 ±16g 时分辨率 1/2048 g/LSB，故原始值除以 2048.0f 得到 g 值。
实验步骤
一、头文件包含与全局定义
包含 imu.h、unistd.h、fcntl.h、termios.h 头文件；声明四个数据更新标志位 flagacc/flaggyro/flagmag/flagangle；定义全量程配置指令数组 SET_FULL_RANGE；实现 check_sum 校验和函数。
#include &lt;unistd.h&gt;
#include &lt;fcntl.h&gt;
#include &lt;termios.h&gt;
#include &lt;string.h&gt;
#include &lt;stdio.h&gt;

// 全局标志位：标记四类数据是否解析完成
static int flagacc = 0, flaggyro = 0, flagmag = 0, flagangle;

// JY901S 指令：开启全量程模式，5 字节指令帧
unsigned char SET_FULL_RANGE[] = {0xFF, 0xAA, 0x23, 0x01, 0x00};

/**
 * @brief 校验和计算函数
 * @param buf 完整 11 字节原始数据缓冲区
 * @return 前 10 字节累加和，用于校验帧完整性
 * @note JY901S 每包数据总长 11 字节，第 11 字节为校验和
 */
static unsigned char check_sum(unsigned char *buf) {
    unsigned char sum = 0;
    for (int i = 0; i &lt; 10; i++) sum += buf[i];
    return sum;
}

二、编写单包解析函数 parse_jy901s
函数接收 11 字节缓存和 IMU_Data 指针。先校验帧头 0x55 与校验和，再通过 switch(buf[1]) 按 CID 分发：0x51 解析加速度、0x52 解析陀螺仪、0x53 解析姿态角、0x54 解析磁力计，每类数据用 buf[低]|buf[高]&lt;&lt;8 拼接 16 位小端值。
/**
 * @brief 解析单包 JY901S 串口原始数据
 * @param buf 11 字节完整帧缓冲区
 * @param imu IMU_Data 结构体指针，存放解析后的原始整型数据
 * @note JY901S 标准帧格式：[0x55][CID][DATA 8字节][CHECKSUM]
 *   帧总长固定 11 字节
 *   buf[0] = 帧头 0x55
 *   buf[1] = 数据类型标识 CID
 *   buf[2]~buf[9] = 三组 16 位小端数据（低字节在前）
 *   buf[10] = 前 10 字节累加校验和
 */
static void parse_jy901s(unsigned char *buf, IMU_Data *imu) {
    // 帧头校验，不是 0x55 直接丢弃
    if (buf[0] != 0x55) return;
    // 校验和校验，校验失败丢弃本帧
    if (check_sum(buf) != buf[10]) return;
    // 存储拼接后的 16 位有符号原始值
    short val;
switch (buf[1]) {  // 根据 CID 区分数据类型
        case 0x51:  // 加速度计 ACC
            // 低字节 buf[2]，高字节 buf[3]，拼接 16 位 short
            val = (short)(buf[2] | (buf[3] &lt;&lt; 8));
            imu-&gt;ax = val;
            val = (short)(buf[4] | (buf[5] &lt;&lt; 8));
            imu-&gt;ay = val;
            val = (short)(buf[6] | (buf[7] &lt;&lt; 8));
            imu-&gt;az = val;
            flagacc = 1;  // 加速度数据已更新标记
            break;
        case 0x52:  // 陀螺仪 GYRO
            val = (short)(buf[2] | (buf[3] &lt;&lt; 8));
            imu-&gt;gx = val;
            val = (short)(buf[4] | (buf[5] &lt;&lt; 8));
            imu-&gt;gy = val;
            val = (short)(buf[6] | (buf[7] &lt;&lt; 8));
            imu-&gt;gz = val;
            flaggyro = 1;  // 陀螺仪数据已更新标记
            break;
        case 0x53:  // 姿态角 ANGLE 横滚/俯仰/偏航
            val = (short)(buf[2] | (buf[3] &lt;&lt; 8));
            imu-&gt;roll = val;
            val = (short)(buf[4] | (buf[5] &lt;&lt; 8));
            imu-&gt;pitch = val;
            val = (short)(buf[6] | (buf[7] &lt;&lt; 8));
            imu-&gt;yaw = val;
            flagangle = 1;  // 姿态角数据已更新标记
            break;
        case 0x54:  // 磁力计 MAG
            val = (short)(buf[2] | (buf[3] &lt;&lt; 8));
            imu-&gt;mx = val;
            val = (short)(buf[4] | (buf[5] &lt;&lt; 8));
            imu-&gt;my = val;
            val = (short)(buf[6] | (buf[7] &lt;&lt; 8));
            imu-&gt;mz = val;
            flagmag = 1;  // 磁力计数据已更新标记
            break;
    }
}

三、串口初始化 imu_init
使用 open 打开串口设备（O_RDWR | O_NOCTTY），tcgetattr 获取当前配置，cfsetispeed/cfsetospeed 设置波特率，配置 8 数据位（CS8）、CLOCAL|CREAD、IGNPAR，关闭行缓冲与输出处理，tcflush 清空缓冲，tcsetattr 生效，最后 write 发送全量程配置指令。
int imu_init(const char *dev_path, int baudrate) {
    // 打开串口，读写模式，不绑定终端
    int fd = open(dev_path, O_RDWR | O_NOCTTY);
    if (fd &lt; 0) return -1;
    struct termios opt;
    tcgetattr(fd, &amp;opt);  // 获取原有串口配置
    // 设置输入输出波特率
    cfsetispeed(&amp;opt, baudrate);
    cfsetospeed(&amp;opt, baudrate);
    opt.c_cflag &amp;= ~CSIZE;
    opt.c_cflag |= CS8 | CLOCAL | CREAD;  // 8 数据位、本地模式、开启接收
    opt.c_iflag = IGNPAR;  // 忽略奇偶校验错误
    opt.c_lflag = 0;       // 关闭行缓冲、回显、终端信号
    opt.c_oflag = 0;       // 关闭输出处理
    tcflush(fd, TCIFLUSH);  // 清空串口接收缓冲区
    tcsetattr(fd, TCSANOW, &amp;opt);  // 立即生效串口配置
    // 发送全量程配置指令，等待设备响应
    write(fd, SET_FULL_RANGE, sizeof(SET_FULL_RANGE));
    return fd;
}

四、状态机组帧与解析 imu_read_parse
定义 11 字节单帧缓存 buf[11] 和静态写指针 ptr。while 循环单字节 read 串口；ptr 为 0 时只有收到 0x55 才开始缓存；凑齐 11 字节后重置 ptr，校验和通过才调用 parse_jy901s；当四类数据全部更新（四个 flag 都置位）后将原始整型值转浮点物理量（加速度除以 2048.0f），清空标志位，返回 1 表示一轮完整数据就绪。
int imu_read_parse(int fd, IMU_Data *imu) {
    static unsigned char buf[11];  // 单帧 11 字节缓存
    static int ptr = 0;            // 缓存写入指针
    while (1) {
        // 单字节读取串口
        if (read(fd, &amp;buf[ptr], 1) &lt;= 0) return -1;
        // 指针归 0 时，只有收到帧头 0x55 才开始缓存
        if (ptr == 0 &amp;&amp; buf[0] != 0x55) continue;
        ptr++;
        // 凑齐完整 11 字节一帧
        if (ptr == 11) {
            ptr = 0;  // 重置指针，准备接收下一帧
            // 校验和正确才解析数据
            if (check_sum(buf) == buf[10])
                parse_jy901s(buf, imu);
            // 四组数据全部更新完毕
            if (flagacc &amp;&amp; flaggyro &amp;&amp; flagmag &amp;&amp; flagangle) {
                // = 原始整型值转实际物理浮点值 =
                // 加速度计：量程 ±16g，分辨率 1/2048 g/LSB
                imu-&gt;ax_f = imu-&gt;ax / 2048.0f;
                imu-&gt;ay_f = imu-&gt;ay / 2048.0f;
                imu-&gt;az_f = imu-&gt;az / 2048.0f;

五、数据打印与主循环
使用 printf 按格式 %6.2f 打印解析后的浮点物理量：加速度（g）、陀螺仪（deg/s）、姿态角（deg）。主程序循环调用 imu_read_parse 读取姿态数据，手动旋转传感器验证读数。
                imu-&gt;gx_f = imu-&gt;gx / 16.0f;
                imu-&gt;gy_f = imu-&gt;gy / 16.0f;
                imu-&gt;gz_f = imu-&gt;gz / 16.0f;
                imu-&gt;roll_f  = imu-&gt;roll * 0.01f;
                imu-&gt;pitch_f = imu-&gt;pitch * 0.01f;
                imu-&gt;yaw_f   = imu-&gt;yaw * 0.01f;
                imu-&gt;mx_f = imu-&gt;mx;
                imu-&gt;my_f = imu-&gt;my;
                imu-&gt;mz_f = imu-&gt;mz;
                // 清空全部更新标记，等待下一轮完整数据
                flagacc = flaggyro = flagmag = flagangle = 0;
                return 1;
            }
        }
    }
}

实验结果

实验结果分析
1、状态机组帧的必要性：JY901S 持续以字节流方式发送 11 字节数据帧，主控 read 的时刻与传感器发送时刻没有同步关系。若一次性 read 11 字节，可能从帧中间开始读取，导致帧头错位、连续多帧解析失败。状态机靠 0x55 帧头自行定位边界，最多丢 1 帧即可重新对齐，是串口协议解析的标准做法。
2、校验和机制保障数据完整性：11 字节帧的第 11 字节为前 10 字节累加和。串口传输可能受电磁干扰出现字节错误，校验失败时直接丢弃本帧不解析，避免错误数据进入控制环节。这是比盲目信任收到的每个字节更稳健的工程策略。
3、16 位小端拼接的原理：JY901S 每个测量值占 2 字节（16 位有符号整数），按小端格式发送（低字节在前、高字节在后），与 ARM/x86 CPU 内部存储顺序一致，无需转换即可直接拼接。代码中 (short)(buf[2]|buf[3]&lt;&lt;8) 的含义：buf[2] 是先收到的低字节直接放最低 8 位，buf[3] 是后收到的高字节左移 8 位挪到高位，两者按位或得到完整 16 位值。
4、四 CID 分发与四 flag 等待机制：加速度(0x51)、陀螺仪(0x52)、姿态角(0x53)、磁力计(0x54)分四帧轮流发送，主控用 switch(buf[1]) 按 CID 分发解析。每解析成功一帧置对应 flag，只有四个 flag 全部置位才算一轮完整数据，此时才转浮点物理量并返回就绪。这避免了加速度已更新但姿态角还是上一轮旧值的混合数据问题。
5、加速度换算系数 1/2048 的由来：JY901S 加速度量程为 ±16g，16 位有符号整数范围 -32768~32767 对应 -16g~+16g，故 1g = 32768/16 = 2048 LSB，即 1 LSB = 1/2048 g。因此原始值除以 2048.0f 得到 g 值。水平静止时重力全部作用在 Z 轴，az_f ≈ 1.0g 符合预期；若静止时三轴均不为 1g，说明传感器安装方向或量程配置有误。
6、常见问题排查：若 read 一直返回 0 或 -1，优先检查串口设备节点是否正确（ls /dev/ttyS*）、权限是否足够（chmod 666）；若读到数据但解析全失败，检查波特率是否与 JY901S 配置一致（9600 或 115200）、TX/RX 是否接反；若校验和频繁失败，检查接线是否松动或电源是否稳定。

实验原理
1、图像预处理原理
（1）灰度化：cvtColor(frame, gray, COLOR_BGR2GRAY)，将三通道彩色图转为单通道灰度图，仅保留亮度信息，减少计算开销。
（2）高斯模糊：GaussianBlur(gray, blur, Size(GZ,GZ), 0)，平滑图像，消除地面灰尘、光斑产生的噪点像素，避免干扰中心点计算。
（3）二值化：通过宏 BINARY_MODE 切换正向 / 反向二值；本实验黑线赛道使用 THRESH_BINARY_INV，黑色轨迹灰度低，转换后像素值 255（白色），地面背景为 0（黑色）。
（4）ROI 裁剪：通过宏 ROI_X1/ROI_Y1/ROI_X2/ROI_Y2 划定画面下方有效区域，过滤上方无关背景，缩小像素遍历范围。
2、道路中心像素遍历计算逻辑
自定义函数 find_RoadCenter(Mat img, int &amp;white_point_count) 双层循环遍历 ROI 内每一个像素：仅累加像素值 = 255 的白色轨迹坐标，统计总白点数量。
道路中心横坐标：center.x = ( Σ 所有白点 x 坐标 ) / white_point_count
道路中心纵坐标：center.y = ( Σ 所有白点 y 坐标 ) / white_point_count
若白点总数小于 WHITE_MAXNUM_STOP，判定赛道丢失，小车立即停机。

示意图说明：红色框为 ROI 裁剪区域，黄色圆点是道路轨迹中心 x₁，绿色圆点是 ROI 理论车头中心 x₂。

4、PID 闭环控制原理
程序内置独立 PID 结构体，包含 Kp、Ki、Kd、输出限幅、积分上下限参数：
（1）初始化阶段：PID_Init() 载入宏定义 KP/KI/KD，限制输出区间 OUT_MIN~OUT_MAX、积分饱和区间；
（2）循环更新：将 IMG_CENTER 设为目标值，道路中心 center.x 作为实际值，传入 PID_Update 计算调节输出；
（3）输出映射：PID 输出值叠加舵机中位 PWM，限制在 SERVO_MIN~SERVO_MAX 区间，通过 PCA9685 修改舵机通道占空比，实现连续转向纠偏；
（4）死区 DEAD_ZONE：微小偏移不输出转向，避免小车直线来回抖动。
5、PCA9685 驱动硬件控制
I2C 通信控制 PCA9685 芯片两路通道：
CH_SERVO (0)：前轮转向舵机，中位 1500，左极限 2000，右极限 1000；
CH_MOTOR (1)：驱动电机，前进固定 1646，停机 1600；封装安全写入函数 pca9685_write_us_safe，防止硬件 PWM 突变冲击。
6、多线程与信号处理
（1）信号捕获：注册 SIGINT 终止信号回调函数，按下 Ctrl+C 时执行停机流程；
（2）互斥锁 pthread_mutex_t：保护 PID 参数、图像数据多线程读写冲突；
（3）UDP 子线程：(可选)开启上位机通信，实时在线修改 Kp/Ki/Kd，观测误差曲线，无需重启程序。

def udp_receiver():
    global data_count, CAR_IP
    sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    sock.bind(("0.0.0.0", RECIVE_PORT))
    print("=" * 60)
    print("PID 实时调参上位机已启动")
    print(f"UDP监听端口 : {RECIVE_PORT}")
    print("等待香橙派小车上传数据...")
    print("=" * 60)
    while True:
        data, addr = sock.recvfrom(2048)
        try:
            msg = data.decode().strip()
            part1, part2 = msg.split("|")
            actual = int(part1.split("=")[1])
            # 解析小车IP，缓存用于下发参数
            if "IP=" in msg:
                CAR_IP = msg.split("IP=")[1].strip()
            # 填充绘图队列
            actual_data.append(actual)
            time_data.append(data_count)
            data_count += 1
            print(f"\r目标:{FIXED_TARGET:3d} | 实际:{actual:3d} | 数据总数:{data_count}  小车IP:{CAR_IP}", end="")
        except Exception:
            # 丢弃格式错误数据包

1、成功完成 USB 摄像头 640×480 画面采集与 ROI 区域裁剪，有效过滤无关背景。

结果分析
1、ROI 裁剪将有效计算区域限制在画面下方 205–405 行，显著减少像素遍历数量，提高实时性。
2、高斯模糊核大小 GZ=7 在降噪与边缘保留之间取得平衡，过大会模糊轨迹边缘，过小则噪点过多影响中心计算。
3、二值化阈值 THRESHOLD=195 是实验关键参数：过低会把地面阴影误判为轨迹，过高会漏掉浅色轨迹，需根据实际赛道光照调整。
4、PID 参数 KP=7.0、KD=3.0 为经验调优值：Kp 太小响应慢、太大则抖动；Kd 有效抑制过冲，但过大会对噪点敏感,Ki不建议使用。
5、WHITE_MAXNUM_STOP=30 作为丢线判定阈值，实际使用中应根据赛道宽度和摄像头高度调整，过低会在弯道误判，过高会延迟响应。

2、USB 摄像头（1~2 路，设备号 0、2）。
3、电脑 MobaXterm 远程连接工具（开启 X11 图像转发）。
4、预装 Python3、OpenCV（opencv-python、opencv-contrib-python）的运行环境。
实验原理
1、摄像头设备检测原理
Linux 下每个 USB 摄像头对应一个 /dev/videoN 设备节点。本机两路摄像头固定为 /dev/video0 和 /dev/video2。程序用 cv2.VideoCapture(i) 依次尝试打开，通过 cap.isOpened() 判断设备是否可用，再读一帧 cap.read() 确认能正常输出图像。每次检测后立即 release 释放设备，加 0.2 秒延时防止设备抢占冲突。
2、FFmpeg 推流服务与摄像头冲突
OrangePi5 上 FFmpeg 推流服务会持续占用摄像头硬件。如果推流服务运行时直接打开摄像头，会出现设备被占用、读取失败。因此程序启动时必须先 systemctl stop 停止推流服务，等待 1 秒让硬件完全释放；程序退出时再 systemctl restart 恢复推流业务。
3、低延迟预览优化
OpenCV 默认的摄像头缓冲区会缓存多帧画面，导致预览出现明显延迟和画面堆积卡顿。通过 cap.set(cv2.CAP_PROP_BUFFERSIZE, 1) 将缓冲区设为仅存 1 帧，每次读取的都是最新帧，消除延迟。同时设置分辨率为 640×480 平衡画质与流畅度。
4、多线程退出机制
OpenCV 的 cv2.waitKey() 按键响应迟钝，在窗口聚焦不稳定时很难捕捉按键。程序启动一个独立的守护线程，用 Python 内置 input() 阻塞等待回车，按下回车后设置全局 exit_flag = True，主循环检测到标志位即退出。守护线程（daemon=True）随主线程结束自动退出，无需手动管理。

import cv2
import time
import os
import threading

# 更新pip
# pip3 install --upgrade pip
# 安装完整版opencv（包含摄像头、图像窗口功能）
# pip3 install opencv-python opencv-contrib-python

# 全局退出标志位，用于线程控制画面循环退出
exit_flag = False

def check_keyboard_exit():
    &quot;&quot;&quot;
    独立线程键盘监听函数
    作用：解决opencv自带waitKey按键响应迟钝问题，按下回车直接标记退出
    &quot;&quot;&quot;
    global exit_flag
    input(&quot;按 **回车键** 退出窗口...\n&quot;)
    exit_flag = True

def get_working_cameras():
    &quot;&quot;&quot;
    遍历检测可用摄像头
    适配硬件：设备两路摄像头固定设备号 0、2
    return: list 存放能正常打开、读取画面的摄像头编号
    &quot;&quot;&quot;
    working = []
    # 只遍历本机实际存在的两路摄像头节点 /dev/video0 /dev/video2
    for i in [0, 2]:
        # 尝试打开摄像头设备
        cap = cv2.VideoCapture(i)
        # 判断设备是否成功打开
        if cap.isOpened():
            # 读取一帧画面，确认摄像头能正常输出图像
            ret, frame = cap.read()
            if ret:
                working.append(i)
        # 关闭摄像头，释放资源，避免占用
        cap.release()
        # 短暂延时，防止设备抢占冲突
        time.sleep(0.2)
    return working

if __name__ == &quot;__main__&quot;:
    # ===================== 前置操作：停止FFmpeg推流服务释放摄像头 =====================
    print(&quot;=== 双摄像头选择工具（完美兼容版）===&quot;)
    print(&quot;[INFO] 正在停止推流服务以释放摄像头...&quot;)
    # 停止主推流服务，屏蔽错误输出
    os.system(&quot;systemctl stop ffmpeg-stream.service 2&gt;/dev/null&quot;)
    # 停止子推流服务
    os.system(&quot;systemctl stop ffmpeg-stream-sub.service 2&gt;/dev/null&quot;)
    # 等待1秒，让服务完全释放摄像头硬件
    time.sleep(1)

    print(&quot;\n正在检测可用摄像头...\n&quot;)

    # 调用函数扫描可用摄像头
    available_cams = get_working_cameras()

    # 无可用摄像头分支
    if not available_cams:
        print(&quot;未找到可用摄像头&quot;)
        print(&quot;[INFO] 恢复推流服务...&quot;)
        os.system(&quot;systemctl restart ffmpeg-stream.service 2&gt;/dev/null&quot;)
        os.system(&quot;systemctl restart ffmpeg-stream-sub.service 2&gt;/dev/null&quot;)
        exit()

    # 打印检测到的摄像头设备号
    print(f&quot;【可用摄像头编号】: {available_cams}&quot;)
    # 手动输入选择要打开的摄像头
    selected = int(input(&quot;请输入要打开的摄像头编号: &quot;))

    # 校验输入合法性
    if selected not in available_cams:
        print(f&quot;只能选择: {available_cams}&quot;)
        print(&quot;[INFO] 恢复推流服务...&quot;)
        os.system(&quot;systemctl restart ffmpeg-stream.service 2&gt;/dev/null&quot;)
        os.system(&quot;systemctl restart ffmpeg-stream-sub.service 2&gt;/dev/null&quot;)
        exit()

    # ===================== 初始化摄像头参数，优化低延迟 =====================
    cap = cv2.VideoCapture(selected)
    # 设置分辨率 宽640 高480
    cap.set(cv2.CAP_PROP_FRAME_WIDTH, 640)
    cap.set(cv2.CAP_PROP_FRAME_HEIGHT, 480)
    # 核心优化：缓冲区仅存1帧，消除画面延迟、堆积卡顿
    cap.set(cv2.CAP_PROP_BUFFERSIZE, 1)

    print(f&quot;\n成功打开摄像头 {selected}&quot;)

    # 启动守护线程，独立监听键盘回车退出
    exit_thread = threading.Thread(target=check_keyboard_exit, daemon=True)
    exit_thread.start()

    # ===================== 画面实时渲染主循环 =====================
    while not exit_flag:
        # 读取一帧图像
        ret, frame = cap.read()
        # 读取失败直接跳出循环
        if not ret:
            print(&quot;读取失败&quot;)
            break

        # 创建窗口并渲染当前帧画面
        cv2.imshow(f&quot;Camera {selected}&quot;, frame)
        # 等待1ms，保证窗口正常刷新，不阻塞画面
        cv2.waitKey(1)

    # ===================== 释放硬件与窗口 =====================
    cap.release()               # 释放摄像头硬件设备
    cv2.destroyAllWindows()     # 销毁所有OpenCV图像窗口

    # 重启之前关闭的FFmpeg推流业务服务，恢复原有业务
    print(&quot;\n[INFO] 程序退出，恢复推流服务...&quot;)
    os.system(&quot;systemctl restart ffmpeg-stream.service 2&gt;/dev/null&quot;)
    os.system(&quot;systemctl restart ffmpeg-stream-sub.service 2&gt;/dev/null&quot;)
    print(&quot;[INFO] 服务已恢复!&quot;)

三、运行程序
在 MobaXterm 终端运行（需开启 X11 转发以显示图像窗口）：
python3 camera_test.py
程序会先停止推流服务，然后检测可用摄像头并打印编号，输入编号后打开预览窗口，按回车退出。
实验结果
1、程序成功检测到可用摄像头编号（如 [0, 2]），并打印供用户选择。
2、成功打开选定摄像头，640×480 分辨率预览画面流畅、无明显延迟。
3、在终端按下回车后，预览窗口立即关闭，摄像头硬件正确释放。
4、程序退出后 FFmpeg 推流服务自动恢复，不影响原有业务。

实验结果分析
1、缓冲区设为 1 帧是低延迟预览的关键：OpenCV 默认缓冲区缓存多帧，导致画面延迟堆积；设为 1 帧后每次读取最新帧，但帧率不稳定时可能出现偶尔跳帧，需根据实际场景权衡。
2、停止 FFmpeg 推流服务是必要前置操作：Linux 下摄像头设备是独占资源，推流服务运行时其他程序无法打开同一设备。等待 1 秒是为了让内核完全释放设备文件描述符，避免打开失败。
3、守护线程 + input() 退出方案比 waitKey 可靠：input() 在终端阻塞等待回车，不依赖 GUI 窗口焦点；daemon=True 保证主线程退出时键盘线程自动结束，不会残留僵尸线程。
4、检测摄像头时每次 release 后加 0.2 秒延时：USB 摄像头释放设备需要硬件层面断电时间，连续打开关闭不加延时会导致后续设备打开失败。

===== 认识香橙派小车实验 =====
实验 认识香橙派小车
实验目的
1、整体认识香橙派智能小车硬件架构，识别小车七大核心硬件组成部件。
2、初步了解每个模块的基础功能，区分各模块外观与安装位置。
3、理清各硬件模块之间的连接关系，知晓基础通信方式。
4、建立小车分层硬件认知，无需进行程序运行、电机控制、传感器读取等实操使用。
实验内容
1、断电状态下观察香橙派小车整机，识别七大硬件单元实物。
2、对照硬件清单，分辨主控、驱动、感知、视觉、语音、电源、网络模块外观特征。
3、观察各模块排线、接口连接方式，区分 I2C、USB、串口线路。
4、梳理各模块基础作用，理解整车硬件布局分工。
5、记录各硬件安装位置，完成小车硬件结构认知。
实验仪器
11、OriginPi 5 主控智能小车整机一台（断电静置状态）
2、配套锂电池
3、硬件识别对照清单
4、电脑一台（仅用于查阅硬件资料，不远程连接小车）
实验原理
1、主控系统（OriginPi 5）：整车核心运算硬件，负责统一接收所有外设采集的数据，完成图像识别、姿态计算、运动指令下发，是小车的 “大脑”。
2、驱动控制（PCA9685 模块）：依靠 I2C 总线和主控相连，接收主控指令输出控制信号，用来控制小车行走电机与视觉云台转向，是小车的 “执行控制器”。
3、环境感知模块
JY901S 九轴姿态传感器：采集小车自身倾斜角度、运动姿态；
GPS 定位模块：获取小车所处地理位置经纬度；
二者配合完成小车自身状态与外部位置感知。
4、视觉系统（双 USB 摄像头）：负责采集外部画面，分别用于赛道识别、障碍物检测，为主控提供视觉原始图像数据。
5、语音交互模块：由 USB 麦克风、音箱组成，麦克风采集外界声音，音箱输出语音，实现人机语音交互硬件基础。
6、电源管理模块：独立稳压硬件，对锂电池电压进行稳定调节，为主控、摄像头、传感器、驱动板等所有硬件提供平稳供电。
7、网络连接（无线网卡）：为小车提供无线网络硬件基础，支持远程连接、画面传输等功能。
实验步骤
小车断电静置，整机外观观察
确认小车电源开关关闭、锂电池不供电，放置在平整桌面，整体观察车身结构，区分车身前部、中部、侧边、尾部硬件安装区域。
二、逐个识别七大硬件模块，记录外观与位置
1主控系统 OriginPi 5
位置：车身中部核心位置，体积最大的主板，带有多组排针、USB 接口。 
2驱动控制 PCA9685 模块
位置：通过细 I2C 排线连接主控，外接云台舵机、车轮电机线路，小型电路板。 
3环境感知硬件 
JY901S 九轴姿态传感器：小型方形模块，安装在车身侧面；
GPS 定位模块：带有天线的方形模组，固定于车身顶部。
4视觉系统 双 USB 摄像头
位置：车头前端一处，云台一处，两根 USB 线接入主控 USB 接口，镜头朝外。 
5语音交互模块 
USB 麦克风：小型拾音头，USB 接口连接主板；
音箱：外置发声喇叭，固定于车头内。
6电源管理稳压模块
位置：电池仓旁，连接锂电池输出线路，分出多路供电线给全车硬件。 
7无线网卡
位置：主板 USB 接口外接，配套外置信号天线。。

三、观察硬件之间的连接线路
1、I2C 细排线：连接 OriginPi 5 主控与 PCA9685 驱动板； 
2、USB 数据线：双摄像头、麦克风、无线网卡统一通过 USB 接口插在主控； 
3、串口排线：连接主控与 JY901S、GPS 感知模块； 
4、供电线束：锂电池接入稳压模块，再由稳压模块分线给所有硬件供电。。
四、整理硬件分层认知，区分各模块作用
按供电层、感知层、核心运算层、执行层、交互层梳理硬件：
供电层：锂电池 + 稳压电源模块；
感知层：双摄像头、JY901S 姿态传感器、GPS 模块、麦克风；
运算核心层：OriginPi 5 主控；
执行驱动层：PCA9685 驱动模块；
交互拓展层：音箱、无线网卡。
五、实验收尾
核对所有硬件均识别完毕，不进行上电、接线、代码运行、电机测试等操作，整理小车配件，完成硬件认识记录。

实验结果
11、能够独立识别香橙派小车全部七大硬件模块，准确说出各部件安装位置与外观特征。
2、清晰区分 I2C 排线、USB 线、串口线、供电线束，掌握各模块基础连接关系。

1、OriginPi 5 作为主控，是所有外设的数据汇总中心，其余硬件均作为外设配合主控工作。
2、PCA9685 依靠 I2C 总线仅用两根线即可控制多路电机、舵机，简化小车内部布线。
3、感知类硬件分为视觉感知、姿态感知、定位感知、声音感知，分别采集不同环境信息。
4、独立稳压模块专门负责电压稳定，保障全车各类硬件可以正常工作，锂电池仅作为储能设备。
5、USB 接口通用性强，摄像头、麦克风、无线网卡都采用 USB 拓展，硬件拆装直观易识别。

===== USB 麦克风实验 =====
USB 麦克风音频采集与播放实验
实验目的
1、理解 Linux 下 USB 声卡、麦克风、扬声器音频设备工作原理，掌握arecord、aplay音频命令使用。
2、掌握 Python 调用系统 shell 命令实现音频录制、音频文件播放的开发方法。
3、学会检测 USB 音频硬件设备、释放被占用声卡资源，解决声卡抢占冲突问题。
4、熟悉单声道、16000Hz 采样率、S16_LE 音频格式等基础音频参数含义。
5、实现自动检测麦克风、5 秒定时录音、录音完成后自动回放音频的完整功能。
实验内容
1、检测系统 USB 声卡，识别匹配型号（A311/Yundea）的麦克风与扬声器硬件。
2、强制清理占用声卡的后台进程、音频服务，释放音频设备避免设备被占用报错。
3、使用arecord命令录制 5 秒单声道 wav 音频文件，采样率固定 16000Hz。
4、判断录音文件存在后，调用aplay通过 USB 声卡扬声器播放录制音频。
5、完善异常捕获逻辑：无声卡、录音失败、播放失败时打印对应提示信息。
6、程序运行结束自动恢复系统音频后台服务。
实验仪器
1、香橙派开发板一台。
2、A311/Yundea 型号 USB 外置声卡（带麦克风 + 扬声器输出）。
3、电脑一台，MobaXterm 远程连接开发板。
4、Python3 运行环境、Linux 内置arecord/aplay音频工具。
实验原理
Linux 音频设备机制
Linux 系统使用 ALSA 音频框架，arecord负责音频录制，aplay负责音频播放；声卡通过cardX,deviceX编号区分硬件，本实验固定使用plughw:3,0作为 USB 声卡通道。
音频基础参数
声道：单声道-c 1，数据量更小，语音识别通用；立体声-c 2多用于音乐。
采样率：16000Hz，语音处理、语音识别标准采样频率。
采样格式：S16_LE，16 位小端 PCM 无损音频格式，wav 文件标准存储格式。
声卡资源抢占问题
pulseaudio、ffmpeg 推流服务、其他录音进程会长期占用声卡，新程序无法读写设备；需要提前杀死占用进程、停止音频服务释放硬件。
Python 调用系统命令
使用subprocess.run()执行 Linux 终端命令，可捕获命令输出、捕获执行异常；os.system()用于执行简单系统操作命令，无返回值捕获。
硬件检测逻辑
通过arecord -l、aplay -l获取设备列表，匹配声卡关键字判断麦克风、扬声器是否正常接入。
实验步骤
一、硬件连接
将 USB 声卡插入香橙派 USB 接口，麦克风接入声卡输入口，耳机 / 音箱接入声卡输出口，远程 MobaXterm 连接开发板。
二、新建实验代码文件
新建mic_audio_test.py，写入完整课程代码，代码内容如下：
import os
import time
import subprocess

RECORD_CHANNELS = 1
TRY_RATES = [16000]
RECORD_SECONDS = 5
RECORD_FILE = "record.wav"
PLAYBACK_FILE = "record.wav"
DEVICE_KEYWORD = ["A311", "Yundea"]

def check_usb_sound_card():
    try:
        arecord_output = subprocess.run(["arecord", "-l"], capture_output=True, text=True).stdout
        aplay_output = subprocess.run(["aplay", "-l"], capture_output=True, text=True).stdout

}

void load_all_templates(const char *folder) {
    DIR *dir = opendir(folder);

float match_all(Mat &amp;frame_small) {
    float max_score = 0;
    for (Mat &amp;tpl : templates) {
        Mat res;

        double val;
        minMaxLoc(res, NULL, &amp;val);

七、实验现象
无标志物：匹配分数低于0.7，电机持续巡航，终端播报“系统启动，开始巡航”。 
摄像头拍到采集的停车标志物：匹配分数≥0.7，播报“检测到停车标志物，立即停车”，电机停止。 
标志物移开：播报“目标消失，恢复巡航”，电机重新行驶
实验结果
1、采集程序camera_save可自动识别摄像头，实时预览画面，按键s成功保存多张标志物图片至 markers 目录。
2、停车程序可自动加载全部采集模板，画面左上角实时打印当前匹配相似度分数。
3、标志物进入画面时匹配分数≥0.7，小车电机立即停止；标志物移出画面，电机自动恢复行驶，播报提示正常输出。
4、任意方式退出程序，电机自动归零停止，PCA9685 资源释放，ffmpeg 推流服务自动恢复，无摄像头 / 硬件占用故障。
结果分析
分程序设计优势
图像采集与识别分离，先批量采集多角度样本再进行匹配，大幅提升识别准确率；若直接单张模板，标志物偏移、缩放后容易识别失败。
模板匹配算法优缺点
优点：实现简单、计算量小、嵌入式香橙派可实时运行；缺点：对标志物旋转、光照剧烈变化鲁棒性差，适合固定场景停车识别。
服务启停必要性
Linux 下同一摄像头只能被一个程序占用，ffmpeg 推流后台常驻会抢占设备，因此采集 / 识别程序启动前必须停止推流，结束后恢复。 
信号捕获安全机制意义
嵌入式小车无人工紧急刹车，若直接强制 kill 程序，电机保持通电会持续行驶；信号捕获保证任何退出场景电机都归零停止，保障硬件安全。

===== 键盘控制无人车实验 =====
键盘控制无人车实验
实验目的
1、理解上位机与下位机基于 TCP/IP 网络通信的远程控制架构。
2、掌握 Python asyncio 异步 TCP 服务器编程，实现多客户端连接管理。
3、掌握 pynput 库键盘事件监听与按键状态去重管理。
4、掌握 JSON 数据格式在网络通信中的封装与解析方法。
5、理解下位机多线程程序设计：主线程接收消息、控制线程执行运动。
6、掌握 PCA9685 驱动电机、舵机，实现键盘控制小车前进、后退、转向。
7、理解按键状态机设计，实现按键按下连续控制、释放自动停车的逻辑。
实验内容
1、PC 端编写 asyncio TCP 服务器，监听 0.0.0.0:8765 端口，等待香橙派连接。
2、上位机使用 pynput 监听键盘按下 / 释放事件，通过 key_state 字典记录按键状态，避免长按重复发送。
3、将按键事件封装为 JSON 消息 {"key": 按键名, "status": press/release}，通过 TCP 发送给香橙派。
4、香橙派下位机连接 PC 服务器，主线程接收并解析按键消息，更新全局按键状态。
5、控制线程根据按键状态调用 PCA9685 控制电机前进 / 后退、舵机左转 / 右转。
6、实现界面显示功能，可视化小车当前运动状态与按键信息。
7、程序退出时安全释放 PCA9685、线程、网络套接字等资源。
实验仪器
1、OrangePi5。
2、PCA9685 I2C 多路 PWM 驱动模块、直流减速电机、转向舵机、配套电源。
3、电脑一台（上位机，预装 Python3、asyncio、pynput 库）。
4、WiFi 或网线网络环境（PC 与香橙派处于同一局域网）。
5、电脑 MobaXterm 远程连接工具（开启 X11 图像转发）。
6、预装 I2C 库、cjson库、pthread 线程库的 C编译环境。
实验原理
1、系统架构总览
本实验采用上位机 + 下位机的 C/S（客户端 / 服务器）架构：
（1）上位机（PC）：运行 Python 程序，作为 TCP 服务器监听 8765 端口，同时通过 pynput 监听键盘事件，将按键状态封装为 JSON 消息转发给已连接的香橙派。
（2）通信链路：基于 TCP/IP 协议，PC 与香橙派处于同一局域网，数据格式为 JSON 字符串 {"key": 按键名, "status": press/release}，以换行符分隔。
（3）下位机（香橙派）：作为 TCP 客户端连接 PC 服务器，主线程接收按键消息并更新状态，控制线程根据状态驱动 PCA9685 控制电机和舵机。

2、上位机异步通信原理
（1）asyncio TCP 服务器：使用 asyncio.start_server(handle_client, PC_HOST, PC_PORT) 启动协程服务器，handle_client 协程处理每个连接的读写与生命周期。
（2）键盘监听线程：pynput.keyboard.Listener 在独立线程中运行，on_press / on_release 回调捕获按键事件，通过 asyncio.run_coroutine_threadsafe 将发送任务投递到主事件循环。
（3）按键状态去重：key_state 字典记录每个按键当前是否处于按下状态，on_press 时若已按下则直接返回，避免操作系统按键重复触发导致消息风暴。
（4）消息广播：send_key 遍历 connected_clients 集合，将 JSON 消息发送给所有已连接客户端，发送失败时自动移除断开的连接。
3、下位机多线程架构
（1）接收线程：作为 TCP 客户端连接 PC 服务器，循环接收数据并按行解析 JSON 消息，更新全局按键状态变量。
（2）控制线程：周期性读取按键状态，根据按键映射调用 PCA9685 写入电机 / 舵机 PWM 数值，实现前进、后退、左转、右转、停止等动作。
（3）界面设置函数：显示小车当前运动状态与按键提示信息。
4、按键控制逻辑
上位机发送按键的 press / release 事件，下位机根据按键状态持续执行对应动作，松开按键时自动停止该方向运动，实现连续平滑控制：
W 键按下：电机前进；松开：电机停机。
S 键按下：电机后退；松开：电机停机。
A 键按下：舵机左转；松开：舵机回中。
D 键按下：舵机右转；松开：舵机回中。
空格键：紧急停车，电机停机、舵机回中。
5、PCA9685 驱动硬件控制
I2C 通信控制 PCA9685 芯片两路通道：CH_SERVO (0) 前轮转向舵机，中位 1500；CH_MOTOR (1) 驱动电机，前进 / 后退 / 停机对应不同 PWM 数值。通过 pca9685_write_us_safe 安全写入函数防止硬件 PWM 突变冲击。

// ===================== TCP 通信线程 =====================
void *thread_tcp(void *arg) {
    int sockfd;
    struct sockaddr_in serv_addr;
    
    while(1) {
        printf("Connecting to %s:%d...\n", PC_SERVER_IP, RECIVE_PORT);
        sockfd = socket(AF_INET, SOCK_STREAM, 0);
        
        if (sockfd &lt; 0) { sleep(1); continue; }
        
        serv_addr.sin_family = AF_INET;
        serv_addr.sin_port = htons(RECIVE_PORT);
        inet_pton(AF_INET, PC_SERVER_IP, &amp;serv_addr.sin_addr);
        
        if (connect(sockfd, (struct sockaddr *)&amp;serv_addr, sizeof(serv_addr)) &lt; 0) {
            close(sockfd); sleep(1); continue;
        }
        
        printf("Connected!\n");  //连接成功打印提示信息
        char buf[4096];
        
        while(1) {
            /* 接收并解析json信息 */
            int n = recv(sockfd, buf, sizeof(buf)-1, 0);
            if (n &lt;= 0) break;
            buf[n] = 0;
            if (g_in_settings_mode) continue;
            
            cJSON *root = cJSON_Parse(buf);
            if (!root) continue;
            cJSON *k = cJSON_GetObjectItem(root, "key");
            cJSON *s = cJSON_GetObjectItem(root, "status");
            if (!k || !s) { cJSON_Delete(root); continue; }
            char c = k-&gt;valuestring[0];
            
            if (!isalpha(c)) { cJSON_Delete(root); continue; }
            int press = !strcmp(s-&gt;valuestring, "press");
            if (c == 'w' || c == 'W') key_w = press;
            if (c == 's' || c == 'S') key_s = press;
            if (c == 'a' || c == 'A') key_a = press;
            if (c == 'd' || c == 'D') key_d = press;
            if (c == 'j' || c == 'J') key_j = press;
            if (c == 'l' || c == 'L') key_l = press;
            if (c == 'i' || c == 'I') key_i = press;
            if (c == 'k' || c == 'K') key_k = press;
            
            if (press) {
                if (c == 'v' || c == 'V') trig_v = 1;
                if (c == 'b' || c == 'B') trig_b = 1;
                if (c == 'n' || c == 'N') trig_n = 1;
                if (c == 'm' || c == 'M') trig_m = 1;
            }
            cJSON_Delete(root);
        }
        
        close(sockfd);
        printf("\nDisconnected\n");
    }
    return NULL;
}

// ===================== 控制线程 =====================
void *thread_ctrl(void *arg) {
    while(1) {
        if (g_in_settings_mode) { usleep(20000); continue; }
        
        if (key_w) cur_esc = FWD;
        else if (key_s) cur_esc = BWD;
        else cur_esc = MOTOR_MID;
        
        if (key_a) cur_steer = L;
        else if (key_d) cur_steer = R;
        else cur_steer = STEER_MID;
        
        if (key_j) pan_pos += PAN_STEP;
        if (key_l) pan_pos -= PAN_STEP;
        if (key_i) tilt_pos += TILT_STEP;
        if (key_k) tilt_pos -= TILT_STEP;
        
        if (pan_pos &lt; SERVO_MIN_US) pan_pos = SERVO_MIN_US;
        if (pan_pos &gt; SERVO_MAX_US) pan_pos = SERVO_MAX_US;
        if (tilt_pos &lt; SERVO_MIN_US) tilt_pos = SERVO_MIN_US;
        if (tilt_pos &gt; SERVO_MAX_US) tilt_pos = SERVO_MAX_US;
        
        pca9685_write_us_safe(CH_Motor, cur_esc);
        pca9685_write_us_safe(CH_TurnServo, cur_steer);
        pca9685_write_us_safe(CH_PAN, pan_pos);
        pca9685_write_us_safe(CH_TILT, tilt_pos);
        
        print_status_line();
        
        if (trig_v) { trig_v = 0; set_motor(); }
        if (trig_b) { trig_b = 0; set_steer(); }
        if (trig_n) { trig_n = 0; set_tilt_step(); }
        if (trig_m) { trig_m = 0; set_pan_step(); }
        
        usleep(10000);
    }
    return NULL;
}

import json

# ===================== 固定配置 =====================
PC_HOST = "0.0.0.0"   # 监听所有网卡（WiFi/网线/热点全都行）

key_state = {}
connected_clients = set()
loop = None
# ===================== 监听TCP客户端处理 =====================
async def handle_client(reader, writer):
    addr = writer.get_extra_info('peername')
    print(f"\n香橙派已连接: {addr}")
    connected_clients.add(writer)
    try:
        while True:
            data = await reader.read(1024)
            if not data:
                break
    except:
        pass
    finally:
        connected_clients.discard(writer)
        writer.close()
        await writer.wait_closed()
        print(f"香橙派断开: {addr}")
# ===================== 发送按键消息 =====================
async def send_key(key_name, status):
    msg = json.dumps({"key": key_name, "status": status}) + "\n"
    print(f"[SEND] {msg.strip()}")
    for writer in list(connected_clients):
        try:
            writer.write(msg.encode())
            await writer.drain()
        except:
            connected_clients.remove(writer)
# ===================== 键盘监听 =====================
def on_press(key):
    try:
        name = key.char.lower()
    except:
        return
    if key_state.get(name, False):
        return
    key_state[name] = True
    asyncio.run_coroutine_threadsafe(send_key(name, "press"), loop)
def on_release(key):
    try:
        name = key.char.lower()
    except:
        return
    key_state[name] = False
    asyncio.run_coroutine_threadsafe(send_key(name, "release"), loop)
# ===================== 主服务 =====================
async def main():
    global loop
    loop = asyncio.get_running_loop()
    # 启动键盘监听
    listener = keyboard.Listener(on_press=on_press, on_release=on_release)
    listener.start()
    # 启动TCP服务（监听所有网卡）
    server = await asyncio.start_server(handle_client, PC_HOST, PC_PORT)
    print("="*50)
    print("PC TCP 服务已启动")
    print(f"监听: 0.0.0.0:8765 (所有网卡)")
    print("香橙派直接连接电脑真实IP即可")
    print("="*50)
    async with server:
        await server.serve_forever()
if __name__ == "__main__":
    asyncio.run(main())

1、成功在 PC 端启动 asyncio TCP 服务器，监听 0.0.0.0:8765 端口，香橙派成功连接。
2、成功通过 pynput 监听键盘按下 / 释放事件，按键状态正确去重，无重复消息发送。
3、成功将按键事件封装为 JSON 消息并通过 TCP 发送，下位机正确接收并解析。
4、下位机主线程正确更新按键状态，控制线程根据状态驱动电机和舵机，小车响应键盘控制。
5、W/S 键控制前进后退、A/D 键控制左右转向、I/K/J/L控制云台、V/B/N/M进行设置，按键释放后自动停止对应动作。
6、界面正确显示小车运动状态与按键提示信息。
7、程序退出时正确释放 PCA9685、线程、网络套接字等全部资源。

结果分析
1、采用 TCP 而非 UDP 通信，保证按键消息的可靠传输，避免丢键导致小车失控；代价是网络抖动时可能有轻微延迟。
2、key_state 字典去重机制有效避免了操作系统按键长按重复触发问题，确保每个按键按下只发送一次 press 消息。
3、上位机监听 0.0.0.0 而非 127.0.0.1，使 WiFi、网线、热点等所有网卡均可接收连接，香橙派只需连接电脑真实 IP。
4、下位机采用主线程接收 + 控制线程执行的双线程架构，消息接收与运动控制解耦，避免网络 I/O 阻塞影响控制实时性。
5、上位机通过 asyncio.run_coroutine_threadsafe 将键盘监听线程的按键事件安全投递到异步事件循环，实现了监听线程与网络 IO 的协作。
6、按键释放即停车的状态机设计，相比按键切换式控制更符合直觉，松手即停提升了操作安全性。
7、若小车响应延迟明显，可检查局域网网络质量、降低控制线程周期、或改用更轻量的消息格式。
